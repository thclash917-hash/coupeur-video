import os import subprocess import shutil import uuid import zipfile import asyncio import time from fastapi import ( FastAPI, File, Form, UploadFile, Header, HTTPException, status, BackgroundTasks, ) from fastapi.responses import FileResponse from fastapi.middleware.cors import CORSMiddleware # ============================================================ # APPLICATION # ============================================================ app = FastAPI( title="Découpeur Vidéo Shorts", version="2.0" ) # ============================================================ # CORS # ============================================================ app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"], ) # ============================================================ # CONFIGURATION # ============================================================ # Mot de passe admin ADMIN_SECRET = os.getenv( "ADMIN_SECRET", "mon_super_secret_123" ) # Dossier principal des tâches BASE_DIR = os.path.abspath( os.getenv("JOBS_DIR", "jobs") ) os.makedirs( BASE_DIR, exist_ok=True ) # ============================================================ # CONFIGURATION DU NETTOYAGE # ============================================================ # Les anciens jobs de plus de X heures peuvent être supprimés # par le bouton admin. # # Exemple : # 6 = suppression des jobs vieux de plus de 6 heures. # # Tu peux modifier cette valeur dans Render avec : # CLEANUP_AGE_HOURS=6 # CLEANUP_AGE_HOURS = float( os.getenv( "CLEANUP_AGE_HOURS", "6" ) ) # ============================================================ # LIMITATION DES TRAITEMENTS SIMULTANÉS # ============================================================ # Nombre maximum de traitements FFmpeg en parallèle. # # 1 = un seul traitement à la fois # 2 = deux utilisateurs peuvent traiter simultanément # 3 = trois utilisateurs... # # Pour Render, 2 est généralement un bon point de départ. # MAX_CONCURRENT_JOBS = int( os.getenv( "MAX_CONCURRENT_JOBS", "2" ) ) FFMPEG_SEMAPHORE = asyncio.Semaphore( MAX_CONCURRENT_JOBS ) # ============================================================ # ROUTE ROOT # ============================================================ @app.get("/") async def root(): return { "status": "online", "message": "Le serveur est bien réveillé !", "max_concurrent_jobs": MAX_CONCURRENT_JOBS, "cleanup_age_hours": CLEANUP_AGE_HOURS, } # ============================================================ # OUTILS JOBS # ============================================================ def get_active_marker(job_dir: str) -> str: """ Retourne le chemin du fichier marqueur indiquant qu'un job est actuellement utilisé. """ return os.path.join( job_dir, ".active" ) def mark_job_active(job_dir: str): """ Marque un job comme actif. Le fichier .active permet au bouton /clean de savoir qu'il ne doit surtout pas supprimer ce dossier. """ marker = get_active_marker( job_dir ) try: with open( marker, "w", encoding="utf-8" ) as f: f.write( str(time.time()) ) except Exception as e: print( f"[JOB] Impossible de créer le marqueur actif : {e}" ) def unmark_job_active(job_dir: str): """ Retire le marqueur actif. """ marker = get_active_marker( job_dir ) try: if os.path.exists(marker): os.remove(marker) except Exception as e: print( f"[JOB] Impossible de supprimer le marqueur actif : {e}" ) def is_job_active(job_dir: str) -> bool: """ Vérifie si un job est actuellement actif. """ marker = get_active_marker( job_dir ) return os.path.exists( marker ) # ============================================================ # NETTOYAGE APRÈS TÉLÉCHARGEMENT # ============================================================ def release_job_after_download(job_dir: str): """ Cette fonction est exécutée par FastAPI après que la réponse FileResponse a terminé son envoi. On retire donc le verrou .active seulement après le téléchargement du ZIP. """ print( f"[JOB] Téléchargement terminé : {job_dir}" ) unmark_job_active( job_dir ) # ============================================================ # FONCTION FFmpeg # ============================================================ def run_ffmpeg(cmd): """ Exécute FFmpeg. Cette fonction est volontairement synchrone. Elle sera appelée dans un thread avec asyncio.to_thread() afin de ne pas bloquer la boucle asyncio de FastAPI. """ print() print("========================================") print("LANCEMENT FFMPEG") print("========================================") print(" ".join(cmd)) print("========================================") print() subprocess.run( cmd, check=True, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=None, ) # ============================================================ # FFmpeg ASYNC # ============================================================ async def run_ffmpeg_async(cmd): """ Lance FFmpeg dans un thread. Cela permet à FastAPI de continuer à répondre aux autres utilisateurs pendant qu'un traitement vidéo est en cours. """ async with FFMPEG_SEMAPHORE: await asyncio.to_thread( run_ffmpeg, cmd ) # ============================================================ # NETTOYAGE D'UN JOB EN CAS D'ERREUR # ============================================================ def delete_job(job_dir: str): """ Supprime complètement un job. """ try: if os.path.exists(job_dir): shutil.rmtree( job_dir, ignore_errors=True ) print( f"[JOB] Job supprimé : {job_dir}" ) except Exception as e: print( f"[JOB] Erreur suppression {job_dir} : {e}" ) # ============================================================ # CUT VIDEO # ============================================================ @app.post("/cut") async def cut_video( file: UploadFile = File(...), segment_duration: int = Form(30), max_clips: int = Form(1), start_min: int = Form(0), end_min: int = Form(10), aspect_ratio: str = Form("original"), quality: str = Form("original"), add_subs: str = Form("false"), background_tasks: BackgroundTasks = None, ): # ======================================================== # 1. VALIDATION # ======================================================== if segment_duration <= 0: raise HTTPException( status_code=400, detail="La durée du segment doit être supérieure à 0." ) if max_clips <= 0: raise HTTPException( status_code=400, detail="Le nombre de clips doit être supérieur à 0." ) if max_clips > 25: raise HTTPException( status_code=400, detail="Le nombre maximum d'extraits est de 25." ) if start_min < 0: raise HTTPException( status_code=400, detail="start_min ne peut pas être négatif." ) if end_min <= start_min: raise HTTPException( status_code=400, detail="end_min doit être supérieur à start_min." ) # ======================================================== # 2. VALIDATION FORMAT # ======================================================== allowed_aspect_ratios = { "original", "9:16", } if aspect_ratio not in allowed_aspect_ratios: raise HTTPException( status_code=400, detail="Format d'image invalide." ) allowed_qualities = { "original", "720p", "1080p", "4k", } if quality not in allowed_qualities: raise HTTPException( status_code=400, detail="Qualité vidéo invalide." ) # ======================================================== # 3. ID UNIQUE # ======================================================== unique_id = uuid.uuid4().hex print() print("========================================") print("NOUVELLE TÂCHE") print("========================================") print(f"JOB ID : {unique_id}") print(f"FICHIER : {file.filename}") print("========================================") print() # ======================================================== # 4. DOSSIER INDIVIDUEL # ======================================================== job_dir = os.path.join( BASE_DIR, unique_id ) os.makedirs( job_dir, exist_ok=False ) # ======================================================== # 5. MARQUER LE JOB COMME ACTIF # ======================================================== mark_job_active( job_dir ) # ======================================================== # 6. NOM DE FICHIER SÉCURISÉ # ======================================================== original_filename = ( file.filename or "video.mp4" ) extension = os.path.splitext( original_filename )[1].lower() if not extension: extension = ".mp4" # On accepte uniquement quelques extensions vidéo allowed_extensions = { ".mp4", ".mov", ".mkv", ".webm", ".avi", ".m4v", } if extension not in allowed_extensions: extension = ".mp4" input_filename = ( f"input{extension}" ) input_path = os.path.join( job_dir, input_filename ) # ======================================================== # 7. UPLOAD # ======================================================== try: with open( input_path, "wb" ) as buffer: while True: chunk = await file.read( 1024 * 1024 ) if not chunk: break buffer.write( chunk ) await file.close() print( f"[JOB {unique_id}] " f"Fichier sauvegardé : {input_path}" ) except Exception as e: print( f"[JOB {unique_id}] " f"Erreur upload : {e}" ) delete_job( job_dir ) raise HTTPException( status_code=500, detail=f"Erreur pendant l'upload : {str(e)}" ) # ======================================================== # 8. PRÉPARATION DES FILTRES # ======================================================== filter_chains = [] # -------------------------------------------------------- # FORMAT 9:16 # -------------------------------------------------------- if aspect_ratio == "9:16": filter_chains.append( "crop=" "min(in_h*9/16\\,in_w):" "min(in_h\\,in_w*16/9):" "(in_w-min(in_h*9/16\\,in_w))/2:" "(in_h-min(in_h\\,in_w*16/9))/2" ) # -------------------------------------------------------- # QUALITÉ # -------------------------------------------------------- if quality == "720p": filter_chains.append( "scale=-2:720" ) elif quality == "1080p": filter_chains.append( "scale=-2:1080" ) elif quality == "4k": filter_chains.append( "scale=-2:2160" ) vf_arg = ( ",".join(filter_chains) if filter_chains else None ) # ======================================================== # 9. GÉNÉRATION DES CLIPS # ======================================================== start_sec = ( start_min * 60 ) output_files = [] for i in range( 1, max_clips + 1 ): current_start_sec = ( start_sec + ((i - 1) * segment_duration) ) output_filename = ( f"extrait_{i}.mp4" ) output_filepath = os.path.join( job_dir, output_filename ) # ==================================================== # COMMANDE FFMPEG # ==================================================== cmd = [ "ffmpeg", "-y", "-hide_banner", "-nostdin", "-i", input_path, "-ss", str(current_start_sec), "-t", str(segment_duration), ] # ==================================================== # FILTRE # ==================================================== if vf_arg: cmd.extend([ "-vf", vf_arg ]) # ==================================================== # ENCODAGE # ==================================================== cmd.extend([ "-r", "30", "-c:v", "libx264", "-preset", "ultrafast", "-crf", "23", "-pix_fmt", "yuv420p", "-c:a", "aac", "-b:a", "128k", "-movflags", "+faststart", output_filepath ]) # ==================================================== # EXÉCUTION # ==================================================== try: print( f"[JOB {unique_id}] " f"Traitement clip {i}/{max_clips}" ) await run_ffmpeg_async( cmd ) if os.path.exists( output_filepath ): output_files.append( ( output_filepath, output_filename ) ) print( f"[JOB {unique_id}] " f"Clip {i} terminé." ) except subprocess.CalledProcessError as e: print( f"[JOB {unique_id}] " f"Erreur FFmpeg clip {i} : {e}" ) # ======================================================== # 10. VÉRIFICATION # ======================================================== if not output_files: delete_job( job_dir ) raise HTTPException( status_code=500, detail="Aucun clip n'a pu être généré." ) # ======================================================== # 11. CRÉATION DU ZIP # ======================================================== zip_filename = ( f"extraits_shorts_{unique_id}.zip" ) zip_path = os.path.join( job_dir, zip_filename ) try: with zipfile.ZipFile( zip_path, "w", compression=zipfile.ZIP_DEFLATED ) as zipf: for ( file_path, arc_name ) in output_files: if os.path.exists( file_path ): zipf.write( file_path, arcname=arc_name ) print( f"[JOB {unique_id}] " f"ZIP créé : {zip_path}" ) except Exception as e: print( f"[JOB {unique_id}] " f"Erreur création ZIP : {e}" ) delete_job( job_dir ) raise HTTPException( status_code=500, detail=f"Erreur création ZIP : {str(e)}" ) # ======================================================== # 12. SUPPRESSION DE LA VIDÉO SOURCE # ======================================================== try: if os.path.exists( input_path ): os.remove( input_path ) except Exception as e: print( f"[JOB {unique_id}] " f"Impossible de supprimer la source : {e}" ) # ======================================================== # 13. VÉRIFICATION DU ZIP # ======================================================== if not os.path.exists( zip_path ): delete_job( job_dir ) raise HTTPException( status_code=500, detail="Le fichier ZIP n'a pas pu être créé." ) # ======================================================== # 14. RÉPONSE # ======================================================== print() print("========================================") print("TÂCHE TERMINÉE") print("========================================") print(f"JOB ID : {unique_id}") print(f"ZIP : {zip_path}") print("========================================") print() # ======================================================== # IMPORTANT : # # On NE retire PAS .active ici. # # Le fichier .active reste présent pendant le téléchargement. # # Il sera retiré par BackgroundTasks seulement après que # FileResponse ait terminé d'envoyer le ZIP. # # Cela empêche /clean de supprimer le ZIP pendant que # l'utilisateur est encore en train de le télécharger. # ======================================================== if background_tasks is not None: background_tasks.add_task( release_job_after_download, job_dir ) return FileResponse( path=zip_path, media_type="application/zip", filename=zip_filename, )
